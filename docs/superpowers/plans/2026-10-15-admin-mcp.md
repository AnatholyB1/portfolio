# MCP admin Sèvalys — Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Claude peut exécuter les actions admin (leads, projets, documents, factures, emails, avis, coûts) via un serveur MCP déployé avec l'app, qui reste synchronisé avec le code.

**Architecture:** Endpoint MCP HTTP `src/app/api/mcp/route.ts` dans l'app Next (chaque déploiement Vercel le met à jour). Auth par token porté en base (`sv_mcp_tokens`, lié à un `sv_admins`), échangé côté serveur contre une session Supabase RLS de cet admin, donc les droits restent ceux de l'admin. La logique de chaque action passe dans `src/lib/server/admin/core/*.ts` (fonctions `(ctx, input)`), la server action devient un wrapper fin, et l'outil MCP appelle le même core. Un test de couverture fait échouer la CI si une `*Action` exportée n'a ni outil MCP ni exclusion motivée.

**Tech Stack:** Next 16 (route handler), `@modelcontextprotocol/sdk`, zod 4, Supabase (service_role pour résoudre le token, session RLS pour agir), vitest.

**Décisions utilisateur (2026-10-15):** déployé dans l'app ; tout est exposé, avec confirmation en deux temps sur l'irréversible (effacement de lead, émission de facture/avoir/document, remboursement, levée de suppression d'e-mail, conversion lead→client).

import 'server-only';
import { createHash, randomBytes } from 'node:crypto';
import { createSupabaseAdminClient } from '@/lib/supabase/admin';

export function generateToken(): string {
  return `svmcp_${randomBytes(32).toString('base64url')}`;
}

export function hashToken(token: string): string {
  return createHash('sha256').update(token).digest('hex');
}

export function isUsable(
  row: { revoked_at: string | null; expires_at: string },
  now: Date = new Date(),
): boolean {
  return row.revoked_at === null && new Date(row.expires_at).getTime() > now.getTime();
}

export type McpPrincipal = { tokenId: string; userId: string };

// Résout « Authorization: Bearer svmcp_… » en admin. Fail-closed : tout échec = null.
export async function resolveToken(authorization: string | null): Promise<McpPrincipal | null> {
  const m = /^Bearer (svmcp_[A-Za-z0-9_-]{43})$/.exec(authorization ?? '');
  if (!m) return null;
  const svc = createSupabaseAdminClient();
  const { data: row } = await svc
    .from('sv_mcp_tokens')
    .select('id, user_id, revoked_at, expires_at')
    .eq('token_hash', hashToken(m[1]))
    .maybeSingle();
  if (!row || !isUsable(row)) return null;
  // Le compte doit être toujours admin au moment de l'appel.
  const { data: admin } = await svc
    .from('sv_admins')
    .select('user_id')
    .eq('user_id', row.user_id)
    .maybeSingle();
  if (!admin) return null;
  await svc.from('sv_mcp_tokens').update({ last_used_at: new Date().toISOString() }).eq('id', row.id);
  return { tokenId: row.id, userId: row.user_id };
}
```

import 'server-only';
import { createClient, type SupabaseClient, type User } from '@supabase/supabase-js';
import { createSupabaseAdminClient } from '@/lib/supabase/admin';
import { getSupabasePublicKey, getSupabaseUrl } from '@/lib/supabase/env';

export type AdminCtx = { supabase: SupabaseClient; user: User };

export async function buildAdminContext(userId: string): Promise<AdminCtx> {
  const svc = createSupabaseAdminClient();
  const { data: u, error: uErr } = await svc.auth.admin.getUserById(userId);
  const email = u?.user?.email;
  if (uErr || !u?.user || !email) throw new Error('mcp_session_failed');

  const { data: link, error: lErr } = await svc.auth.admin.generateLink({ type: 'magiclink', email });
  const hashed = link?.properties?.hashed_token;
  if (lErr || !hashed) throw new Error('mcp_session_failed');

  const anon = createClient(getSupabaseUrl(), getSupabasePublicKey(), {
    auth: { persistSession: false, autoRefreshToken: false },
  });
  const { data: sess, error: vErr } = await anon.auth.verifyOtp({ token_hash: hashed, type: 'magiclink' });
  const accessToken = sess?.session?.access_token;
  if (vErr || !accessToken) throw new Error('mcp_session_failed');

  const supabase = createClient(getSupabaseUrl(), getSupabasePublicKey(), {
    auth: { persistSession: false, autoRefreshToken: false },
    global: { headers: { Authorization: `Bearer ${accessToken}` } },
  });
  return { supabase, user: u.user };
}

import 'server-only';
import { z } from 'zod';
import type { AdminCtx } from './session';
import type { McpPrincipal } from './token';

export type AuditEntry = { tokenId: string; userId: string; tool: string; destructive: boolean; outcome: 'ok' | 'error' | 'needs_confirmation' };

export type ToolDef<S extends z.ZodObject = z.ZodObject> = {
  name: string;
  description: string;
  /** Noms des server actions que cet outil remplace (lu par coverage.test.ts). */
  covers: string[];
  destructive?: boolean;
  input: S;
  preview?: (ctx: AdminCtx, input: z.infer<S>) => Promise<unknown>;
  run: (ctx: AdminCtx, input: z.infer<S>) => Promise<unknown>;
};

export function defineTool<S extends z.ZodObject>(def: ToolDef<S>): ToolDef<S> {
  return def;
}

export type ToolResult =
  | { ok: true; data: unknown }
  | { ok: true; needsConfirmation: true; preview: unknown }
  | { ok: false; error: string };

export async function runTool<S extends z.ZodObject>(
  tool: ToolDef<S>,
  raw: unknown,
  ctx: AdminCtx,
  principal: McpPrincipal,
  audit: (e: AuditEntry) => Promise<void> | void,
): Promise<ToolResult> {
  const base = { tokenId: principal.tokenId, userId: principal.userId, tool: tool.name, destructive: Boolean(tool.destructive) };
  // `confirm` est ajouté aux outils destructifs, jamais aux autres.
  const schema = tool.destructive ? tool.input.extend({ confirm: z.literal(true).optional() }) : tool.input;
  const parsed = schema.safeParse(raw);
  if (!parsed.success) {
    await audit({ ...base, outcome: 'error' });
    return { ok: false, error: 'invalid_input' };
  }
  const { confirm, ...input } = parsed.data as Record<string, unknown> & { confirm?: true };
  try {
    if (tool.destructive && confirm !== true) {
      const preview = tool.preview ? await tool.preview(ctx, input as z.infer<S>) : { input };
      await audit({ ...base, outcome: 'needs_confirmation' });
      return { ok: true, needsConfirmation: true, preview };
    }
    const data = await tool.run(ctx, input as z.infer<S>);
    await audit({ ...base, outcome: 'ok' });
    return { ok: true, data };
  } catch {
    console.error(`[mcp] ${tool.name} failed`);
    await audit({ ...base, outcome: 'error' });
    return { ok: false, error: 'tool_failed' };
  }
}
```

// src/lib/server/mcp/index.ts
import 'server-only';
import { McpServer } from '@modelcontextprotocol/sdk/server/mcp.js';
import { leadTools } from './tools/leads';
import { runTool, type AuditEntry, type ToolDef } from './registry';
import type { AdminCtx } from './session';
import type { McpPrincipal } from './token';
import { createSupabaseAdminClient } from '@/lib/supabase/admin';

export const ALL_TOOLS: ToolDef[] = [...leadTools];

async function audit(e: AuditEntry) {
  await createSupabaseAdminClient().from('sv_mcp_audit').insert({
    token_id: e.tokenId, user_id: e.userId, tool: e.tool, destructive: e.destructive, outcome: e.outcome,
  });
}

export function buildServer(ctx: AdminCtx, principal: McpPrincipal): McpServer {
  const server = new McpServer({ name: 'sevalys-admin', version: '1.0.0' });
  for (const tool of ALL_TOOLS) {
    const shape = (tool.destructive ? tool.input.extend({ confirm: z.literal(true).optional() }) : tool.input).shape;
    server.registerTool(tool.name, { description: tool.description, inputSchema: shape }, async (args: unknown) => {
      const result = await runTool(tool, args, ctx, principal, audit);
      return { content: [{ type: 'text' as const, text: JSON.stringify(result) }], isError: !result.ok };
    });
  }
  return server;
}

import { WebStandardStreamableHTTPServerTransport } from '@modelcontextprotocol/sdk/server/webStandardStreamableHttp.js';
import { NextResponse } from 'next/server';
import { buildServer } from '@/lib/server/mcp';
import { buildAdminContext } from '@/lib/server/mcp/session';

export const dynamic = 'force-dynamic';
export const maxDuration = 60;

async function handle(request: Request): Promise<Response> {
  const principal = await resolveToken(request.headers.get('authorization'));
  if (!principal) return NextResponse.json({ error: 'unauthorized' }, { status: 401 });
  const ctx = await buildAdminContext(principal.userId);
  const server = buildServer(ctx, principal);
  const transport = new WebStandardStreamableHTTPServerTransport({ sessionIdGenerator: undefined });
  await server.connect(transport);
  return transport.handleRequest(request);
}

- rules.ts: MAIL_EVENTS = ['client_invited','step_changed','onboarding_completed'] as const; type MailEvent; type MailTemplate = 'invite' | 'step_changed' | 'onboarding_completed'; type Rule = { template: MailTemplate; delayMs: number; to: 'client' | 'admin' }; MAIL_RULES: Record<MailEvent, Rule> (all delayMs 0); ADMIN_NOTIFY_EMAIL = 'contact@sevalys.com'; dedupeKey: { clientInvited(clientId, email, resendN?: number), stepChanged(factId, email), onboardingCompleted(projectId, email) } producing `client_invited:{clientId}:{email}` (+ `:resend:{n}`), `step_changed:{factId}:{email}`, `onboarding_completed:{projectId}:{email}`, emails lowercased, result <= 256 chars.
- stepChangedEmail.ts: buildStepChangedEmail({ stepName, expectedAction, portalUrl }) -> subject `Votre projet avance : ${stepName}`, CTA "Ouvrir mon espace".

  - enqueueMail({ event, recipientEmail, dedupeKey, payload, clientId?, projectId? }): Promise<{ id: string | null; inserted: boolean }> — upsert with { onConflict: 'dedupe_key', ignoreDuplicates: true } then select id; send_after = now + rule.delayMs; template/recipient_kind from MAIL_RULES.
  - sendOutboxRow(id): Promise<'sent' | 'failed' | 'not_claimed'> — claim via update set status 'sending', attempts+1 (read attempts in the returned row), claimed_at now where id = id and status in ('pending','failed') and send_after <= now, returning the row; then deliver.
  - deliver(row) (internal): buildMail(row) -> resend.emails.send({ from, to, replyTo, subject, html, text }, { idempotencyKey: row.dedupe_key }); success -> status 'sent', sent_at, provider_id; error -> status 'failed', last_error short code ('resend_error' | 'no_api_key' | 'render_error' | 'exception').
  - enqueueAndSend(input): Promise<'sent' | 'pending' | 'failed' | 'duplicate'> — never throws.
  - processDueMail(limit = 25): Promise<{ claimed: number; sent: number; failed: number }> — callRpc('mail/outbox', 'sv_claim_due_mail', { p_limit: limit }) then deliver each.
  - buildMail(row): { from, replyTo, subject, html, text } per template (invite uses buildInviteEmail with payload.clientName and buildLoginUrl(recipient_email); step_changed uses payload.stepName/expectedAction and buildPortalUrl(); onboarding_completed uses payload.companyName/projectTitle and buildAdminProjectUrl(project_id)).
